> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Port rlmlocal-site's code-grounded perception primitives into RGI so the engine builds a multi-language structural graph (AST nodes, import edges, call edges) before any recursive spawning happens.

**Architecture:** Add a new `rgi/perception/rlmlocal_compat/` package that mirrors rlmlocal's `StructureExtractor` and `importGraph` modules in Python using `tree-sitter` bindings. Replace/extend RGI's current `PerceptionLayer.ingest_codebase()` to build a `CognitiveGraph` from the richer substrate. Keep the existing RGI orchestration unchanged; only the world-model it reasons over improves.

**Tech Stack:** Python 3.12+, `tree-sitter` + `tree-sitter-python`/`tree-sitter-javascript`/`tree-sitter-typescript`/`tree-sitter-go`/`tree-sitter-rust`, pytest, existing RGI `CognitiveGraph` models.

from dataclasses import dataclass


@dataclass(frozen=True)
class LanguagePack:
    family: str
    grammar_module: str
    extensions: tuple[str, ...]
    test_file_regex: str
    call_expression_node_type: str
    function_like_node_types: tuple[str, ...]
    method_container_types: tuple[str, ...]


LANGUAGE_PACKS: dict[str, LanguagePack] = {
    "python": LanguagePack(
        family="python",
        grammar_module="tree_sitter_python",
        extensions=(".py",),
        test_file_regex=r"(_test|test_)\.py$|tests?/.*\.py$",
        call_expression_node_type="call",
        function_like_node_types=("function_definition", "lambda"),
        method_container_types=("class_definition",),
    ),
    "js": LanguagePack(
        family="js",
        grammar_module="tree_sitter_javascript",
        extensions=(".js", ".jsx", ".ts", ".tsx", ".mjs"),
        test_file_regex=r"\.(test|spec)\.(ts|tsx|js|jsx|mjs)$",
        call_expression_node_type="call_expression",
        function_like_node_types=("function_declaration", "function_expression", "arrow_function", "method_definition"),
        method_container_types=("class_declaration", "object"),
    ),
    "go": LanguagePack(
        family="go",
        grammar_module="tree_sitter_go",
        extensions=(".go",),
        test_file_regex=r"_test\.go$",
        call_expression_node_type="call_expression",
        function_like_node_types=("function_declaration", "func_literal", "method_declaration"),
        method_container_types=("type_declaration",),
    ),
    "rust": LanguagePack(
        family="rust",
        grammar_module="tree_sitter_rust",
        extensions=(".rs",),
        test_file_regex=r"#\[cfg(test)\]",
        call_expression_node_type="call_expression",
        function_like_node_types=("function_item", "closure_expression"),
        method_container_types=("impl_item", "trait_item"),
    ),
}

_EXTENSION_TO_PACK: dict[str, LanguagePack] = {
    ext: pack for pack in LANGUAGE_PACKS.values() for ext in pack.extensions
}


def lang_family(ext: str) -> str | None:
    pack = _EXTENSION_TO_PACK.get(ext.lower())
    return pack.family if pack else None


def supported_extensions() -> set[str]:
    return set(_EXTENSION_TO_PACK.keys())
```

"""Lazy tree-sitter parser loader ported from rlmlocal's treeSitter.ts."""
import importlib
from pathlib import Path

from tree_sitter import Language, Parser

from rgi.perception.rlmlocal_compat.language_packs import LanguagePack, lang_family


_PARSER_CACHE: dict[str, Parser] = {}


def load_parser(pack: LanguagePack) -> Parser:
    if pack.family in _PARSER_CACHE:
        return _PARSER_CACHE[pack.family]
    grammar = importlib.import_module(pack.grammar_module)
    language = Language(grammar.language())
    parser = Parser(language)
    _PARSER_CACHE[pack.family] = parser
    return parser


def parse_file(path: Path) -> Parser | None:
    pack = lang_family(path.suffix)
    if pack is None:
        return None
    parser = load_parser(LANGUAGE_PACKS[pack])
    return parser.parse(path.read_bytes())
```

"""Port of rlmlocal's StructureExtractor.ts to Python."""
from dataclasses import dataclass, field
from pathlib import Path

from rgi.perception.rlmlocal_compat.tree_sitter_loader import parse_file


@dataclass
class CodeStructure:
    functions: list[dict] = field(default_factory=list)
    classes: list[dict] = field(default_factory=list)
    methods: list[dict] = field(default_factory=list)
    imports: list[dict] = field(default_factory=list)
    calls: list[dict] = field(default_factory=list)
    call_sites: list[dict] = field(default_factory=list)


def _node_text(node, source: bytes) -> str:
    return source[node.start_byte:node.end_byte].decode("utf-8", errors="replace")


def _enclosing_name(node, source: bytes) -> str | None:
    parent = node.parent
    while parent:
        if parent.type == "class_definition":
            for child in parent.children:
                if child.type == "identifier":
                    return _node_text(child, source)
        parent = parent.parent
    return None


def extract_structure(path: Path) -> CodeStructure:
    tree = parse_file(path)
    if tree is None:
        return CodeStructure()
    source = path.read_bytes()
    struct = CodeStructure()

    for node in tree.root_node.children:
        if node.type == "function_definition":
            name = next((c for c in node.children if c.type == "identifier"), None)
            if name:
                struct.functions.append({
                    "name": _node_text(name, source),
                    "line": node.start_point[0] + 1,
                    "span": (node.start_byte, node.end_byte),
                })
        elif node.type == "class_definition":
            name = next((c for c in node.children if c.type == "identifier"), None)
            if name:
                struct.classes.append({
                    "name": _node_text(name, source),
                    "line": node.start_point[0] + 1,
                    "span": (node.start_byte, node.end_byte),
                })
            for child in node.children:
                if child.type == "function_definition":
                    mname = next((c for c in child.children if c.type == "identifier"), None)
                    if mname:
                        struct.methods.append({
                            "name": _node_text(mname, source),
                            "class": _node_text(name, source),
                            "line": child.start_point[0] + 1,
                            "span": (child.start_byte, child.end_byte),
                        })
        elif node.type == "import_statement":
            for child in node.children:
                if child.type == "dotted_name":
                    struct.imports.append({
                        "name": _node_text(child, source),
                        "line": node.start_point[0] + 1,
                        "is_from": False,
                    })
                elif child.type == "aliased_import":
                    for sub in child.children:
                        if sub.type == "dotted_name":
                            struct.imports.append({
                                "name": _node_text(sub, source),
                                "line": node.start_point[0] + 1,
                                "is_from": False,
                            })
        elif node.type == "import_from_statement":
            module = next((c for c in node.children if c.type == "dotted_name"), None)
            names = [c for c in node.children if c.type == "import_list"]
            if module and names:
                for name_node in names[0].children:
                    if name_node.type in ("identifier", "aliased_import"):
                        struct.imports.append({
                            "name": _node_text(name_node, source).split(" ")[0],
                            "module": _node_text(module, source),
                            "line": node.start_point[0] + 1,
                            "is_from": True,
                        })

    # Call sites: simple callee name extraction
    for node in tree.root_node.descendants:
        if node.type == "call":
            func = node.child_by_field_name("function")
            if func:
                struct.calls.append({
                    "callee": _node_text(func, source),
                    "line": node.start_point[0] + 1,
                    "span": (node.start_byte, node.end_byte),
                })

    return struct
```

"""Port of rlmlocal's importGraph.ts core edge builder."""
from dataclasses import dataclass, field
from pathlib import Path

from rgi.perception.rlmlocal_compat.structure_extractor import CodeStructure


@dataclass
class ImportGraph:
    edges: list[dict] = field(default_factory=list)
    reverse_edges: dict[str, list[dict]] = field(default_factory=dict)
    symbol_defs: dict[str, list[dict]] = field(default_factory=dict)


def build_import_graph(root: Path, structs: dict[Path, CodeStructure]) -> ImportGraph:
    graph = ImportGraph()
    files_by_stem = {p.stem: p for p in structs.keys()}

    # Index symbol definitions
    for path, struct in structs.items():
        for fn in struct.functions:
            graph.symbol_defs.setdefault(fn["name"], []).append({"file": path, "kind": "function"})
        for cls in struct.classes:
            graph.symbol_defs.setdefault(cls["name"], []).append({"file": path, "kind": "class"})
        for method in struct.methods:
            graph.symbol_defs.setdefault(method["name"], []).append({
                "file": path, "kind": "method", "class": method.get("class"),
            })

    # Build import edges
    for path, struct in structs.items():
        for imp in struct.imports:
            if imp.get("is_from"):
                module = imp.get("module", "")
                target = files_by_stem.get(module.split(".")[0])
                if target:
                    edge = {
                        "source_file": str(path),
                        "target_file": str(target),
                        "symbol": imp["name"],
                        "module": module,
                        "line": imp.get("line"),
                    }
                    graph.edges.append(edge)
                    graph.reverse_edges.setdefault(str(target), []).append(edge)
            else:
                top = imp["name"].split(".")[0]
                target = files_by_stem.get(top)
                if target:
                    edge = {
                        "source_file": str(path),
                        "target_file": str(target),
                        "symbol": imp["name"],
                        "line": imp.get("line"),
                    }
                    graph.edges.append(edge)
                    graph.reverse_edges.setdefault(str(target), []).append(edge)

    return graph
```

"""RGI perception layer backed by rlmlocal-style structural extraction."""
from pathlib import Path

from rgi.core.models import CognitiveEdge, CognitiveGraph, CognitiveNode, GraphPolicy, GraphState, LoopType, NodeType
from rgi.perception.rlmlocal_compat.import_graph import ImportGraph, build_import_graph
from rgi.perception.rlmlocal_compat.structure_extractor import extract_structure


class RlmlocalPerceptionLayer:
    def ingest_codebase(self, path: str) -> CognitiveGraph:
        root = Path(path)
        graph = CognitiveGraph(
            loop_type=LoopType.KNOWLEDGE,
            state=GraphState(objective=f"World model for {root}"),
            policy=GraphPolicy(auto_spawn=False, require_verification=False),
        )

        py_files = sorted(root.rglob("*.py"))
        structs = {p: extract_structure(p) for p in py_files}
        import_graph = build_import_graph(root, structs)

        file_to_node: dict[Path, str] = {}
        for py_file in py_files:
            node = CognitiveNode(
                type=NodeType.MEMORY,
                content=f"Module {py_file.stem} in {py_file.name}",
                confidence=1.0,
                parent_graph_id=graph.id,
                metadata={"file": str(py_file), "name": py_file.stem, "entity_kind": "module"},
            )
            graph.nodes[node.id] = node
            file_to_node[py_file] = node.id

        for py_file, struct in structs.items():
            for fn in struct.functions:
                node = CognitiveNode(
                    type=NodeType.MEMORY,
                    content=f"Function {fn['name']} in {py_file.name}",
                    confidence=1.0,
                    parent_graph_id=graph.id,
                    metadata={
                        "file": str(py_file), "name": fn["name"], "entity_kind": "function",
                        "line": fn.get("line"),
                    },
                )
                graph.nodes[node.id] = node
                graph.edges.append(CognitiveEdge(
                    source=file_to_node[py_file], target=node.id,
                    edge_type="contains", weight=1.0,
                ))

        for edge in import_graph.edges:
            src = Path(edge["source_file"])
            tgt = Path(edge["target_file"])
            if src in file_to_node and tgt in file_to_node:
                graph.edges.append(CognitiveEdge(
                    source=file_to_node[src], target=file_to_node[tgt],
                    edge_type="imports", weight=0.9,
                    metadata={"symbol": edge.get("symbol"), "line": edge.get("line")},
                ))

        return graph
```

"""Grounded exploration primitives ported from rlmlocal's gate/tool patterns."""
import re
from pathlib import Path


def read_file(params: dict) -> dict:
    path = Path(params["path"])
    line_start = params.get("line_start")
    line_end = params.get("line_end")
    lines = path.read_text().splitlines()
    if line_start is not None and line_end is not None:
        snippet = "\n".join(lines[line_start - 1:line_end])
    else:
        snippet = path.read_text()
    return {"findings": [{"file": str(path), "content": snippet}], "confidence": 1.0}


def grep(params: dict) -> dict:
    pattern = re.compile(params["pattern"])
    root = Path(params["root"])
    glob = params.get("glob", "*.py")
    findings = []
    for path in root.rglob(glob):
        for i, line in enumerate(path.read_text().splitlines(), start=1):
            if pattern.search(line):
                findings.append({"file": str(path), "line": i, "content": line})
    return {"findings": findings, "confidence": 1.0}


def callers(params: dict) -> dict:
    symbol = params["symbol"]
    edges = params.get("import_graph_edges", [])
    results = [e for e in edges if e.get("symbol") == symbol]
    return {"findings": results, "confidence": 0.9}
```
